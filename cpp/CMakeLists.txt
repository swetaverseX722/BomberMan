cmake_minimum_required(VERSION 3.16)
project(bomberman_cpp LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(CURSES_NEED_NCURSES TRUE)
find_package(Curses REQUIRED)

add_executable(bomberman src/main.cpp)
target_include_directories(bomberman PRIVATE ${CURSES_INCLUDE_DIR})
target_link_libraries(bomberman PRIVATE ${CURSES_LIBRARIES})

option(BOMBERMAN_BUILD_SDL_GRAPHICS "Build cross-platform SDL2 graphics target" ON)
option(BOMBERMAN_BUILD_COCOA_GRAPHICS "Build macOS Cocoa graphics target" ON)

if(BOMBERMAN_BUILD_SDL_GRAPHICS)
  find_package(SDL2 QUIET CONFIG)

  if(TARGET SDL2::SDL2)
    add_executable(bomberman_graphics_sdl src/graphics_sdl.cpp)
    target_link_libraries(bomberman_graphics_sdl PRIVATE SDL2::SDL2)
    if(TARGET SDL2::SDL2main)
      target_link_libraries(bomberman_graphics_sdl PRIVATE SDL2::SDL2main)
    endif()

    find_package(SDL2_image QUIET CONFIG)
    if(TARGET SDL2_image::SDL2_image)
      target_link_libraries(bomberman_graphics_sdl PRIVATE SDL2_image::SDL2_image)
      target_compile_definitions(bomberman_graphics_sdl PRIVATE BOMBERMAN_USE_SDL_IMAGE=1)
    endif()
  else()
    find_package(SDL2 QUIET)
    if(SDL2_FOUND)
      add_executable(bomberman_graphics_sdl src/graphics_sdl.cpp)
      target_include_directories(bomberman_graphics_sdl PRIVATE ${SDL2_INCLUDE_DIRS})
      target_link_libraries(bomberman_graphics_sdl PRIVATE ${SDL2_LIBRARIES})

      find_package(SDL2_image QUIET)
      if(SDL2_IMAGE_FOUND OR SDL2_image_FOUND)
        if(DEFINED SDL2_IMAGE_INCLUDE_DIRS)
          target_include_directories(bomberman_graphics_sdl PRIVATE ${SDL2_IMAGE_INCLUDE_DIRS})
        endif()
        if(DEFINED SDL2_IMAGE_LIBRARIES)
          target_link_libraries(bomberman_graphics_sdl PRIVATE ${SDL2_IMAGE_LIBRARIES})
        endif()
        target_compile_definitions(bomberman_graphics_sdl PRIVATE BOMBERMAN_USE_SDL_IMAGE=1)
      endif()
    else()
      message(WARNING "SDL2 not found: skipping bomberman_graphics_sdl target. Install SDL2 to enable cross-platform graphics.")
    endif()
  endif()
endif()

if(APPLE AND BOMBERMAN_BUILD_COCOA_GRAPHICS)
  enable_language(OBJCXX)
  add_executable(bomberman_graphics src/graphics_main.mm)
  target_compile_options(bomberman_graphics PRIVATE -fobjc-arc)
  target_link_libraries(bomberman_graphics PRIVATE "-framework Cocoa")
endif()
